Scratch test runner
#!/bin/bash
set -eu -o pipefail
scratch=$(cd "$(dirname "$0")" && pwd -P)
label=$1
shift
stamp=$(date -u '+%Y%m%dT%H%M%SZ')
basetemp="$scratch/pytest-$label-$stamp"
log="$scratch/$label.txt"
export TMPDIR="$scratch"
export UV_CACHE_DIR="$scratch/uv-cache"
export UV_OFFLINE=1
export PYTHONDONTWRITEBYTECODE=1
export PYTHONPATH="$scratch"
{
    printf 'Started %s UTC\n' "$stamp"
    df -H /
    available=$(df -k / | awk 'NR==2 {print $4}')
    if [ "$available" -lt 58593750 ]; then
        printf '%s\n' 'Below the 60 GB free-disk limit.'
        exit 70
    fi
    if [[ "$stamp" > 20261125T235035Z ]]; then
        printf '%s\n' 'Past the new-test start deadline.'
        exit 71
    fi
    command=(uv run --no-sync --with 'pytest-xdist>=3.6' pytest -n 4 -p no:cacheprovider
        -p round4_isolation "--basetemp=$basetemp" -q -rs "$@")
    printf 'Command'
    printf ' %q' "${command[@]}"
    printf '\n'
    result=0
    "${command[@]}" || result=$?
    printf 'Finished %s, exit %s\n' "$(date -u '+%Y-%m-%d %H:%M:%S UTC')" "$result"
    du -sh "$scratch"
    exit "$result"
} 2>&1 | tee "$log"
