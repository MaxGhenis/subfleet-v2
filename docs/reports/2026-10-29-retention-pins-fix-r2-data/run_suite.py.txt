import ast
import concurrent.futures
import json
import os
from pathlib import Path
import subprocess
import sys

scratch = Path('/private/tmp/sf-fix272r2')
plan = [('green-review-r2', ['tests/fake/test_retention_dependency_review_r2.py', 'tests/unit/test_retention_alternates_review_r2.py'])]
whole = [
 'tests/fake/test_retention_dependency_review_r1.py',
 'tests/fake/test_retention_dependency_incomplete.py',
 'tests/unit/test_retention_alternates.py',
 'tests/unit/test_retention_job_dependencies.py',
 'tests/fake/test_retention_dependencies.py',
 'tests/fake/test_retention_dependency_spelling.py',
 'tests/unit/test_retention_dependency_units.py',
 'tests/unit/test_retention_folded_pin.py',
 'tests/fake/test_writer_retention_fence.py',
]
for file in whole:
    plan.append((Path(file).stem, [file]))
for file, group_size in [
 ('tests/unit/test_retention_dependency_properties.py', 1),
 ('tests/unit/test_retention_shared_folders.py', 4),
 ('tests/fake/test_detached_retention_fence.py', 4),
]:
    names = [node.name for node in ast.parse(Path(file).read_text()).body
             if isinstance(node, ast.FunctionDef) and node.name.startswith('test_')]
    for index in range(0, len(names), group_size):
        plan.append((f'{Path(file).stem}-{index // group_size + 1}',
                     [file, '-k', ' or '.join(names[index:index+group_size])]))
for file in ['tests/fake/test_state_contract.py', 'tests/unit/test_test_module_names.py', 'tests/unit/test_invariants_index.py']:
    plan.append((Path(file).stem, [file]))
(scratch/'suite-plan.json').write_text(json.dumps(plan, indent=2)+'\n')

def run(item):
    name, args = item
    tmp = scratch / ('hold-' + name)
    tmp.mkdir()
    env = dict(os.environ, TMPDIR=str(tmp), HYPOTHESIS_STORAGE_DIRECTORY=str(scratch/'hyp'), PYTHONDONTWRITEBYTECODE='1')
    cmd = ['/bin/zsh', '/Users/maxghenis/reviews/subfleet-hub/host-runs/suite_hold.sh',
           'env', '-u', 'SUBFLEET_RETENTION_DORMANT', '-u', 'PYTHONPATH',
           f'UV_PROJECT_ENVIRONMENT={scratch}/venv', f'UV_CACHE_DIR={scratch}/uv-cache',
           'uv', 'run', '--no-sync', 'python', str(scratch/'run_hold.py'), name, *args]
    with (scratch/(name+'.log')).open('w') as log:
        result = subprocess.run(cmd, env=env, stdout=log, stderr=subprocess.STDOUT)
    print(f'{name}: rc={result.returncode}', flush=True)
    return name, result.returncode

with concurrent.futures.ThreadPoolExecutor(max_workers=2) as pool:
    results = list(pool.map(run, plan))
(scratch/'suite-results.json').write_text(json.dumps(results, indent=2)+'\n')
raise SystemExit(any(code for _, code in results))
