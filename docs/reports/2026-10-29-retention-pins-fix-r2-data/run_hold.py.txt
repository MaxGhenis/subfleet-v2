import collections
import datetime
import json
import os
from pathlib import Path
import resource
import signal
import sys
import time

import pytest

scratch = Path('/private/tmp/sf-fix272r2')
name, *args = sys.argv[1:]
# Fence assertions include rendered paths; keep fixture roots free of the
# word 'retention', independent of the descriptive hold name.
import tempfile
fixture_tmp = scratch / ('cases-' + str(os.getpid()))
fixture_tmp.mkdir()
os.environ['TMPDIR'] = str(fixture_tmp)
tempfile.tempdir = None
started = datetime.datetime.now(datetime.timezone.utc).isoformat()
loads = os.getloadavg()
clock = time.monotonic()
data = {'name': name, 'start': started, 'load': loads, 'args': args, 'results': [], 'collected': []}

class Observe:
    def pytest_collection_finish(self, session):
        data['collected'] = [item.nodeid for item in session.items]
    def pytest_runtest_logreport(self, report):
        if report.when == 'call' or report.failed or (report.skipped and report.when == 'setup'):
            data['results'].append({'node': report.nodeid, 'phase': report.when, 'outcome': report.outcome, 'xfail': getattr(report, 'wasxfail', None)})

signal.alarm(570)
print(f'OBSERVED_START {name} {started} load={loads}', flush=True)
try:
    code = pytest.main(['-q', '-p', 'no:cacheprovider', '--hypothesis-seed=272', *args], plugins=[Observe()])
finally:
    signal.alarm(0)
    data['seconds'] = time.monotonic() - clock
    data['max_rss_bytes'] = resource.getrusage(resource.RUSAGE_SELF).ru_maxrss
    data['end'] = datetime.datetime.now(datetime.timezone.utc).isoformat()
    (scratch / f'{name}.json').write_text(json.dumps(data, indent=2) + '\n')
print(f'OBSERVED_END {name} seconds={data["seconds"]:.2f}', flush=True)
raise SystemExit(code)
