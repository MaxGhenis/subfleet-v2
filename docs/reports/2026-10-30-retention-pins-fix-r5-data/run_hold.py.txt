    definition = json.loads(Path('/private/tmp/sf-fix272r5/definitions.json').read_text())[mutant.removeprefix('mutant-').removesuffix('-final')]
    class Loader(importlib.machinery.SourceFileLoader):
        def get_code(self, fullname):
            source = Path(self.get_filename(fullname)).read_text()
            mutated = source
            baseline = source
            for index, patch in enumerate(definition.get('patches', [definition]), 1):
                assert mutated.count(patch['old']) == 1
                mutated = mutated.replace(patch['old'], patch['new'])
                if index == definition.get('baseline_patch_count', 0):
                    baseline = mutated
            loaded.append(dict(module=fullname, original_sha256=hashlib.sha256(source.encode()).hexdigest(), baseline_sha256=hashlib.sha256(baseline.encode()).hexdigest(), mutated_sha256=hashlib.sha256(mutated.encode()).hexdigest()))
            return compile(mutated, self.get_filename(fullname), 'exec')
    class Finder(importlib.abc.MetaPathFinder):
        def find_spec(self, fullname, path=None, target=None):
            if fullname != definition['file'].removesuffix('.py').replace('/', '.'):
                return None
            spec = importlib.machinery.PathFinder.find_spec(fullname, path)
            spec.loader = Loader(fullname, spec.origin)
            return spec
    sys.meta_path.insert(0, Finder())

    @pytest.hookimpl(tryfirst=True)
    def pytest_collection_modifyitems(self, session, config, items):
        all_nodes.extend(item.nodeid for item in items)

        phases.setdefault(report.nodeid, {})[report.when] = report.outcome
