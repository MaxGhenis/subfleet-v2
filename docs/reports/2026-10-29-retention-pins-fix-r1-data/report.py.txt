def counts(data):
    tally=collections.Counter()
    for node in data['collected']:
        p=data['phases'].get(node,{})
        if all(p.get(k)=='passed' for k in ('setup','call','teardown')):tally['P']+=1
        elif 'failed' in p.values() and p.get('teardown')=='passed':tally['F']+=1
        elif 'skipped' in p.values():tally['S']+=1
        else:tally['U']+=1
    return '/'.join(str(tally[k]) for k in ('P','F','S','U'))
