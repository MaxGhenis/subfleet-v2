"""One old/new retention pin-query comparison on a synthetic, scratch-owned store.

Ported from the pending 2026-10-06 measurement. Exactly one timed read per side;
no repetitions or other scale runs. The old SQL is from release/217 c9430763f.
The new read mirrors the final _pin_reasons owner query and dependency loop.
Run under suite_hold.sh: python tools/measure_retention_pins.py OUT JOBS LIVE.
Only JOBS/LIVE pairs 3000/100 and 6000/600 are supported.
"""
from __future__ import annotations
import argparse,datetime,hashlib,json,os,platform,random,sqlite3,sys,tempfile,time
from pathlib import Path
sys.path.insert(0,str(Path(__file__).resolve().parents[1]))
from subfleet import dependencies,retention
from subfleet.store import Store
OLD_IN_USE = ("SELECT a.job_id FROM jobs a JOIN jobs b ON b.job_id <> a.job_id "
              "AND b.state NOT IN ('succeeded','failed','cancelled','lost') "
              "AND (b.worktree = a.worktree COLLATE NOCASE OR b.workdir = a.worktree COLLATE NOCASE "
              "OR b.workdir LIKE a.worktree || '/%' OR b.worktree LIKE a.worktree || '/%') "
              "WHERE a.worktree IS NOT NULL AND a.in_place = 0 AND a.sandbox = 'workspace-write'")

OLD_UNRECORDED = ("SELECT a.job_id FROM jobs a JOIN jobs b ON b.job_id <> a.job_id "
                  "AND b.state NOT IN ('succeeded','failed','cancelled','lost') "
                  "AND (b.worktree = (?1 || a.job_id) COLLATE NOCASE OR b.workdir = (?1 || a.job_id) COLLATE NOCASE "
                  "OR b.workdir LIKE ?1 || a.job_id || '/%' OR b.worktree LIKE ?1 || a.job_id || '/%') "
                  "WHERE a.worktree IS NULL AND a.in_place = 0 AND a.sandbox = 'workspace-write'")

ROOT = "/Users/x/.subfleet"

def build(path: Path, *, owners: int, live: int, events_per_job: int, long_waiters: int, long_events: int,
          legacy: int, exports: int, seed: int = 7) -> Store:
    rng = random.Random(seed)
    store = Store(path)
    conn = store.connection
    conn.execute("BEGIN IMMEDIATE")
    now = "2026-10-06T00:00:00Z"
    rows, events = [], []
    for n in range(owners):
        job_id = f"20261001-{n:06d}-owner"
        rows.append((job_id, f"req-{job_id}", "d", "dispatch", "succeeded", f"/Users/x/src/repo{n % 40}",
                     f"{ROOT}/worktrees/{job_id}" if n % 9 else None, "/p", "workspace-write", 0, now))
        events.append((now, "job.submitted", job_id, json.dumps({"folder": f"/Users/x/src/repo{n % 40}",
                                                                 "depends": {"git": [f"/Users/x/src/repo{n % 40}/.git"]}})))
        events.extend((now, "job.waiting", job_id, "{}") for _ in range(events_per_job))
    owner_ids = [row[0] for row in rows]
    for n in range(live):
        job_id = f"20261006-{n:06d}-live"
        inside = rng.random() < 0.2
        target = rng.choice(owner_ids)
        workdir = f"{ROOT}/worktrees/{target}/sub" if inside else f"/Users/x/src/repo{n % 40}"
        rows.append((job_id, f"req-{job_id}", "d", "dispatch", rng.choice(["queued", "waiting", "running"]),
                     workdir, None, "/p", "read-only", 0, now))
        submitted = {} if n < legacy else {
            "folder": workdir, "depends": {"git": [f"/Users/x/src/repo{n % 40}/.git"],
                                           **({"out_path": f"/Users/x/out/{n}.md"} if n % 3 == 0 else {}),
                                           **({"review_root": f"/Users/x/src/repo{n % 40}"} if n % 5 == 0 else {})}}
        events.append((now, "job.submitted", job_id, json.dumps(submitted)))
        count = long_events if n < long_waiters else events_per_job
        events.extend((now, "job.waiting", job_id, "{}") for _ in range(count))
    conn.executemany("INSERT INTO jobs(job_id,request_id,payload_digest,kind,state,workdir,worktree,prompt_path,"
                     "sandbox,in_place,created_at) VALUES (?,?,?,?,?,?,?,?,?,?,?)", rows)
    conn.executemany("INSERT INTO events(ts,kind,job_id,data_json) VALUES (?,?,?,?)", events)
    for n in range(exports):
        conn.execute("INSERT INTO leases(lease_key,holder,acquired_at) VALUES (?,?,?)",
                     (f"out:/Users/x/out/e{n}.md", owner_ids[n], now))
    conn.commit()
    conn.execute("ANALYZE")
    return store

def old_pin_query(store: Store) -> set[str]:
    return {row["job_id"] for row in [*store.query(OLD_IN_USE),
            *store.query(OLD_UNRECORDED, (f"{ROOT}/worktrees/",))]}


def new_pin_query(store: Store) -> set[str]:
    jobs = store.query("SELECT job_id,kind,state,finished_at,worktree,sandbox,in_place FROM jobs")
    needs = dependencies.users(dependencies.live(store.query))
    found = set()
    for row in jobs:
        if row["sandbox"] != "workspace-write" or row["in_place"]:
            continue
        tree = row["worktree"] or str(Path(ROOT) / "worktrees" / row["job_id"])
        if tree and needs.get(dependencies.key(tree), set()) - {row["job_id"]}:
            found.add(row["job_id"])
    return found


def main() -> None:
    parser=argparse.ArgumentParser(description=__doc__)
    parser.add_argument("out",type=Path);parser.add_argument("jobs",type=int);parser.add_argument("live",type=int)
    args=parser.parse_args()
    if (args.jobs,args.live) not in ((3000,100),(6000,600)):
        parser.error("choose 3000/100 or 6000/600")
    params=dict(owners=args.jobs-args.live,live=args.live,events_per_job=40,
                long_waiters=5 if args.jobs==3000 else 20,long_events=5000,legacy=0,
                exports=10 if args.jobs==3000 else 50)
    report=dict(start=datetime.datetime.now(datetime.timezone.utc).isoformat(),load=os.getloadavg(),
                jobs=args.jobs,live=args.live,parameters=params,python=sys.version.split()[0],sqlite=sqlite3.sqlite_version,
                host=platform.node(),source_sha256={p:hashlib.sha256(Path(p).read_bytes()).hexdigest()
                  for p in ('subfleet/dependencies.py','subfleet/retention.py')})
    with tempfile.TemporaryDirectory(prefix="pin-measure-",dir=os.environ["TMPDIR"]) as directory:
        store=build(Path(directory)/"synthetic.sqlite3",**params)
        try:
            report['events']=store.one('SELECT count(*) AS n FROM events')['n']
            with store.transaction('measure.pin-reads'):
                report['load_at_queries']=os.getloadavg()
                start=time.perf_counter();old=old_pin_query(store);report['old_ms']=(time.perf_counter()-start)*1000
                report['load_at_new_query']=os.getloadavg()
                start=time.perf_counter();new=new_pin_query(store);report['new_ms']=(time.perf_counter()-start)*1000
            assert old==new,(sorted(old-new),sorted(new-old))
            # These fixtures add outside-tree dependencies and exports: the pin sets
            # should agree, even though the new query covers more roles in general.
            report['pins']=len(new);report['ratio']=report['new_ms']/report['old_ms']
            report['plans']={name:[r['detail'] for r in store.query('EXPLAIN QUERY PLAN '+sql,values)]
                for name,sql,values in [('old recorded',OLD_IN_USE,()),('old unrecorded',OLD_UNRECORDED,(f'{ROOT}/worktrees/',)),
                                       ('new live',dependencies._LIVE,()),('new exports',dependencies._EXPORTING,()),
                                       ('new owners','SELECT job_id,kind,state,finished_at,worktree,sandbox,in_place FROM jobs',())]}
        finally:store.close()
    report['load_end']=os.getloadavg()
    args.out.write_text(json.dumps(report,indent=2)+'\n');print(json.dumps(report,indent=2))


if __name__=='__main__':main()
