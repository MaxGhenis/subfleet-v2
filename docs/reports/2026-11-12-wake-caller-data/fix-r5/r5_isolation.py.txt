"""Keep round 4's pytest evidence inside its own scratch directory."""
from pathlib import Path
import os
import shlex
import sys
import tempfile
import pytest
from tests import failed_keep

failed_keep.ROOT = Path(__file__).resolve().parent.parent / "failed-state"


# Socket fixtures explicitly use /tmp to fit macOS's Unix socket path limit.
# The canonical path must be short because the daemon resolves it before bind.
fixture_roots = os.environ.get("FIX5_FIXTURE_ROOTS")
if fixture_roots:
    roots = Path(fixture_roots).resolve()
    assert roots.is_dir() and roots.name == "sf-wcall-fix5"
    mkdtemp = tempfile.mkdtemp

    def scratch_mkdtemp(suffix=None, prefix=None, dir=None):
        if dir in ("/tmp", "/private/tmp"):
            dir = fixture_roots
        return mkdtemp(suffix, prefix, dir)

    tempfile.mkdtemp = scratch_mkdtemp



def forbidden_execution(event, args):
    if event == 'subprocess.Popen':
        command = args[1]
    elif event in ('os.exec', 'os.posix_spawn', 'os.spawn'):
        command = args[1]
    elif event == 'os.system':
        command = args[0]
    else:
        return
    if isinstance(command, bytes):
        command = os.fsdecode(command)
    try:
        words = shlex.split(command) if isinstance(command, str) else [os.fsdecode(word) for word in command]
    except (TypeError, ValueError):
        return
    if len(words) >= 3 and Path(words[0]).name in ('sh', 'bash', 'zsh') and '-c' in words:
        try:
            words = shlex.split(words[words.index('-c') + 1])
        except ValueError:
            return
    names = [Path(word).name for word in words]
    if any(name in ('agent-secret', 'security', 'launchctl', 'ps', 'lsof') for name in names):
        pytest.skip('round-5 brief prohibits this fixture command or live process inspection')
    for index, name in enumerate(names):
        tail = words[index + 1:]
        if name == 'git' and tail[:1] == ['push']:
            pytest.skip('round-5 brief prohibits executing git push, including fixtures')
        if name != 'gh':
            continue
        read = (
            tail[:2] in (['pr', 'view'], ['pr', 'list'], ['pr', 'checks'], ['pr', 'diff'],
                         ['run', 'list'], ['run', 'view'], ['issue', 'view'], ['issue', 'list'],
                         ['repo', 'view'])
            or tail[:1] == ['api'] and read_only_api(tail[1:])
        )
        if not read:
            pytest.skip('round-5 brief permits only read-only gh commands')


def read_only_api(args):
    for index, word in enumerate(args):
        if word in ('-X', '--method'):
            if args[index + 1:index + 2] != ['GET']:
                return False
        elif word.startswith(('-f', '-F', '--field', '--raw-field', '--input', '--method=', '-X')):
            if word not in ('--method=GET', '-XGET'):
                return False
    return True


def pytest_collection_modifyitems(items):
    for item in items:
        if item.nodeid.startswith('tests/unit/test_compat.py::test_python_m_subfleet_runs_a_hook_end_to_end'):
            item.add_marker(pytest.mark.skip(reason='SessionStart launches a detached worker'))


def pytest_terminal_summary(terminalreporter):
    for report in terminalreporter.stats.get('skipped', ()):
        reason = report.longrepr[2]
        terminalreporter.write_line(f'SKIPPED {report.nodeid}: {reason}')



sys.addaudithook(forbidden_execution)
