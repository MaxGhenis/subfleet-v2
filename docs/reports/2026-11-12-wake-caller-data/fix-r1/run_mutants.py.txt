def sha(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def main() -> int:
    tree, python, scratch, out = (Path(sys.argv[1]), sys.argv[2], Path(sys.argv[3]), Path(sys.argv[4]))
    wanted = set(sys.argv[5:])
    results = json.loads(out.read_text()) if out.exists() else []
    env = {key: value for key, value in os.environ.items() if key != "PYTHONPATH"}
    env.update(PYTHONDONTWRITEBYTECODE="1", HYPOTHESIS_STORAGE_DIRECTORY=str(scratch / "hyp"))
    for mutant in MUTANTS:
        if wanted and mutant["name"] not in wanted:
            continue
        path = tree / mutant["file"]
        pristine = path.read_bytes()
        digest = sha(pristine)
        text = pristine.decode()
        replacements = mutant.get("replacements") or [(mutant["old"], mutant["new"])]
        mutated = text
        for old, new in replacements:
            found = mutated.count(old)
            expected = mutant.get("count", 1)
            assert found >= 1 and (expected is None or found == expected), (mutant["name"], old, found)
            mutated = mutated.replace(old, new)
        started = time.time()
        load = os.getloadavg()[0]
        try:
            path.write_text(mutated)
            run = subprocess.run(
                [python, "-m", "pytest", "-q", "-p", "no:cacheprovider", "-x",
                 f"--basetemp={scratch / ('pytest-mutant-' + mutant['name'])}", *mutant["killers"]],
                cwd=tree, env=env, capture_output=True, text=True, timeout=540)
            rc, tail = run.returncode, (run.stdout + run.stderr).strip().splitlines()[-3:]
        except subprocess.TimeoutExpired:
            rc, tail = None, ["timeout after 540 s"]
        finally:
            path.write_bytes(pristine)
        restored = sha(path.read_bytes())
        assert restored == digest, (mutant["name"], "restore failed")
        results.append({"name": mutant["name"], "finding": mutant["finding"], "file": mutant["file"],
                        "killers": mutant["killers"], "rc": rc, "killed": rc == 1,
                        "summary": tail, "seconds": round(time.time() - started, 1),
                        "load_at_start": load, "sha256_before": digest, "sha256_after_restore": restored})
        out.write_text(json.dumps(results, indent=1))
        print(mutant["name"], "killed" if rc == 1 else f"SURVIVED rc={rc}", tail[-1] if tail else "", flush=True)
    return 0


if __name__ == "__main__":
    sys.exit(main())
