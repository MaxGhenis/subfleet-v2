    class Loader(importlib.machinery.SourceFileLoader):
        def get_code(self, fullname):
            source = Path(self.get_filename(fullname)).read_text()
            mutated = source
            baseline = source
            for index, patch in enumerate(definition.get('patches', [definition]), 1):
                assert mutated.count(patch['old']) == 1
                mutated = mutated.replace(patch['old'], patch['new'])
                if index == definition.get('baseline_patch_count', 0):
                    baseline = mutated
            loaded.append(dict(module=fullname, original_sha256=hashlib.sha256(source.encode()).hexdigest(), baseline_sha256=hashlib.sha256(baseline.encode()).hexdigest(), mutated_sha256=hashlib.sha256(mutated.encode()).hexdigest()))
            return compile(mutated, self.get_filename(fullname), 'exec')
    class Finder(importlib.abc.MetaPathFinder):
        def find_spec(self, fullname, path=None, target=None):
            if fullname != definition['file'].removesuffix('.py').replace('/', '.'):
                return None
            spec = importlib.machinery.PathFinder.find_spec(fullname, path)
            spec.loader = Loader(fullname, spec.origin)
            return spec
    sys.meta_path.insert(0, Finder())

class Observer:
    @pytest.hookimpl(tryfirst=True)
    def pytest_collection_modifyitems(self, session, config, items):
        all_nodes.extend(item.nodeid for item in items)
    def pytest_sessionfinish(self, session, exitstatus):
        from subfleet.home_paths import PYTEST_SENTINEL
        base = session.config._subfleet_safe_home_base
        assert (base / PYTEST_SENTINEL).is_file(), "scratch home marker removed by the observer's basetemp"
    def pytest_collection_finish(self, session):
        nodes.extend(item.nodeid for item in session.items)
        for item in session.items:
            if label.startswith("shared-folders-400-"):
                # This property's settings decorator is defined inside the test.
                # Split only phases; retain its full 400 generated examples.
                original_settings = item.module.settings
                def phase_settings(*args, _item=item, **kwargs):
                    configured = original_settings(*args, **kwargs)
                    if configured.max_examples != 400:
                        return configured
                    from hypothesis import settings as hs, Phase, Verbosity
                    phases = ((Phase.explicit,) if label.endswith("-explicit") else
                              tuple(p for p in configured.phases if p != Phase.explicit))
                    selected = hs(parent=configured, phases=phases, verbosity=Verbosity.verbose)
                    def decorate(fn):
                        result = selected(fn)
                        property_settings[_item.nodeid] = dict(max_examples=selected.max_examples,
                            derandomize=selected.derandomize, database_none=selected.database is None,
                            phases=[str(p) for p in selected.phases],
                            explicit_examples=len(getattr(result, "hypothesis_explicit_examples", [])))
                        return result
                    return decorate
                item.module.settings = phase_settings
            configured = getattr(item.obj, "_hypothesis_internal_use_settings", None)
            if configured is not None and (label.startswith("retention_sweep_race-") or label.startswith("salvage-split-")):
                from hypothesis import settings as hypothesis_settings, Phase, Verbosity
                selected = ((Phase.explicit,) if label.endswith("-explicit") else tuple(p for p in configured.phases if p != Phase.explicit))
                configured = hypothesis_settings(parent=configured, phases=selected, verbosity=Verbosity.verbose)
                item.obj._hypothesis_internal_use_settings = configured
            if configured is not None:
                property_settings[item.nodeid] = {"max_examples": configured.max_examples, "derandomize": configured.derandomize, "phases": [str(p) for p in configured.phases], "explicit_examples": len(getattr(item.obj, "hypothesis_explicit_examples", []))}
    def pytest_runtest_logreport(self, report):
