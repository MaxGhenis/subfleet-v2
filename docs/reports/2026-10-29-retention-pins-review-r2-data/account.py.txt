s = Path(__file__).parent

plan = json.loads((s / "ordinary-plan.json").read_text())

observations = []
for path in s.glob("*.json"):
    data = json.loads(path.read_text())
    if isinstance(data, dict) and "phases" in data:

        log = s / (path.stem + ".log")

def count(obs):
    counts = collections.Counter()
    for node in obs["collected"]:
        phase = obs["phases"].get(node, {})
        if "failed" in phase.values():
            counts["F"] += 1
        elif phase.get("call") == "skipped" and "test_retention_dependency_review_r2.py" in node:
            counts["X"] += 1
        elif "skipped" in phase.values():
            counts["S"] += 1
        elif phase.get("call") == "passed" and phase.get("teardown") == "passed":
            counts["P"] += 1
        else:
            counts["U"] += 1
    return counts

def tally(c):
    return "/".join(str(c.get(k, 0)) for k in ("P", "F", "S", "X", "U"))

ordinary = [v for v in observations if v["name"] in names]
assert all(v["exitstatus"] == 0 for v in ordinary)
assert all(count(v)["U"] == 0 and count(v)["F"] == 0 for v in ordinary)
latest = {}
first = {}
for obs in ordinary:
    for node in obs["collected"]:
        # The full final new-file run supersedes the unparameterized cycle's older run.
        latest[node] = obs
        first.setdefault(node.split("::")[0], obs)
files = collections.defaultdict(list)
for node, obs in latest.items():
    files[node.split("::")[0]].append((node, obs))

file_counts = {}
file_lines = ["| File | P/F/S/X/U | First observed start UTC | Load (1/5/15 min) |",
              "|---|---|---|---|"]
for file, items in sorted(files.items()):
    counts = collections.Counter()
    for node, obs in items:
        slim = dict(obs, collected=[node])
        counts.update(count(slim))
    file_counts[file] = dict(counts)
    initial = first[file]
    file_lines.append(f"| §{file}§ | {tally(counts)} | {initial['start'][11:23]} | "
                      + " / ".join(f"{n:.2f}" for n in initial["load"]) + " |")
total = collections.Counter()
for c in file_counts.values():
    total.update(c)
assert sum(1 for node in latest if node.startswith("tests/") and
           "review_r2" not in node) == 543

holds = []
longest = (0.0, "")
for obs in observations:
    log = (s / (obs["name"] + ".log")).read_text()
    durations = re.findall(r" in ([0-9.]+)s(?: |$)", log, re.MULTILINE)
    elapsed = float(durations[-1]) if durations else None
    if elapsed is not None:
        longest = max(longest, (elapsed, obs["name"]))
        assert elapsed < 600, (obs["name"], elapsed)
    slot = re.search(r"hold \S+ slot ([AB])", log)
    counts = count(obs)
    holds.append(dict(name=obs["name"], start=obs["start"], load=obs["load"],
                      counts=dict(counts), selected=len(obs["collected"]),
                      slot=slot.group(1) if slot else None, pytest_seconds=elapsed,
                      exitstatus=obs["exitstatus"], mutant=obs.get("mutant")))

mutants = [v for v in observations if v["name"].startswith("mutant-")]
assert len(mutants) == 18
mutant_lines = ["| Named mutant | Result / observed killing node | P/F/S/X/U | Start UTC | Load (1 min) |",
                "|---|---|---|---|---|"]
for obs in mutants:
    assert obs["exitstatus"] == 1 and len(obs["loaded"]) == 1
    failed = [n for n, phases in obs["phases"].items() if "failed" in phases.values()]
    assert failed, obs["name"]
    mutant_lines.append(f"| §{obs['mutant']}§ | KILLED: §{failed[0]}§ | {tally(count(obs))} | "
                        + obs["start"][11:23] + f" | {obs['load'][0]:.2f} |")

hold_lines = ["| Hold | Start UTC | Load (1/5/15 min) | P/F/S/X/U | Slot | Pytest seconds |",

                      + f" | {tally(h['counts'])} | {h['slot'] or '-'} | {h['pytest_seconds']} |")

for prop in properties:
